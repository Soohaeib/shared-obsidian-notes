mat_available = mat_beg + net_purchases
dm_used = mat_available - mat_end

# Factory Overhead (Manufacturing Overhead):
# Other Factory Expenses + Factory Insurance + Depreciation - Machinery

# Total Manufacturing Costs Incurred:
total_mfg_costs = dm_used + direct_labor + factory_oh

# Total Cost of Work in Process to Account for:
total_wip_account = wip_beg + total_mfg_costs

# Cost of Goods Manufactured:
cogm = total_wip_account - wip_end

cogs = fg_available - fg_end

# (c) Statement of Comprehensive Income (Income Statement)
# Net Sales:
net_sales = sales - sales_discounts

# Gross Profit / Gross Margin:
gross_profit = net_sales - cogs

# Selling Expenses:
# Sales Salaries + Freight Out
selling_expenses = sales_salaries + freight_out

# Total Operating Expenses:
operating_expenses = selling_expenses + admin_exp

# Net Income / Operating Income:
net_income = gross_profit - operating_expenses

print("Net Sales:", net_sales)
