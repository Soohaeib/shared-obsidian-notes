# Normal rate = $15/hr
# Overtime up to 9 hours/day = Single rate ($15/hr)
# Overtime over 9 hours/day = Double rate ($30/hr)

normal_hrs_daily = 0
ot_single_daily = 0
ot_double_daily = 0

for h in hours_b:
    if h <= 8:
        normal_hrs_daily += h
    elif h == 9:
        normal_hrs_daily += 8
        ot_single_daily += 1
    else: # h > 9
        normal_hrs_daily += 8
        ot_single_daily += 1 # 9th hour at single rate
        ot_double_daily += (h - 9) # hours above 9 at double rate

earnings_daily = (normal_hrs_daily * 15) + (ot_single_daily * 15) + (ot_double_daily * 30)

print(f"Daily Basis:")
print(f" Normal hours (up to 8/day): {normal_hrs_daily}")
