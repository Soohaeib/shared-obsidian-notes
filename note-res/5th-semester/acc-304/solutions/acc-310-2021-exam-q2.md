# Question 2 (a)
# J. Martin employed by Beach City Canning Company
# Monday: 270 units, 8 hours
# Tuesday: 210 units, 8 hours
# Wednesday: 300 units, 8 hours
# Thursday: 240 units, 8 hours
# Friday: 260 units, 8 hours
# Total units = 270 + 210 + 300 + 240 + 260 = 1280
# Total hours = 40 hours
# Base wage = $15 per hour
# Standard production rate = 30 units per hour
# Standard daily production = 8 hours * 30 units/hr = 240 units
# 100 percent bonus plan:
# Under 100% bonus plan (standard hour plan / group / individual):
# Efficiency % = Actual Units / Standard Units (or Actual Production per hour / Standard Production per hour)
# If Efficiency > 100%, Earnings = Actual Units * Standard Unit Rate (or Base Pay * Efficiency)
# Standard Unit Rate = Base Wage / Standard Rate per hour = $15 / 30 = $0.50 per unit.
# If Efficiency <= 100%, Guaranteed Base Wage = 8 hrs * $15 = $120 per day.

daily_data_a = [
    ('Monday', 270, 8),
    ('Tuesday', 210, 8),
    ('Wednesday', 300, 8),
    ('Thursday', 240, 8),
    ('Friday', 260, 8)
]

print("--- Question 2(a) ---")
total_units_a = 0
total_hours_a = 0
total_earnings_a = 0

for day, units, hrs in daily_data_a:
    std_units = hrs * 30 # 240
    eff = (units / std_units) * 100 # efficiency %
    std_rate = 15.0 / 30.0 # $0.50 per unit
    
    # 100% Bonus plan daily earnings:
    if units > std_units:
        earnings = units * std_rate # or hrs * 15 * (eff / 100)
    else:
        earnings = hrs * 15.0 # Guaranteed base wage
        
    eff_hourly_rate = earnings / hrs
    labor_cost_per_unit = earnings / units
    
    total_units_a += units
    total_hours_a += hrs
    total_earnings_a += earnings
    
    print(f"{day}: Units={units}, Std={std_units}, Eff={eff:.2f}%, Earnings=${earnings:.2f}, Eff Hourly Rate=${eff_hourly_rate:.3f}, Cost/Unit=${labor_cost_per_unit:.3f}")

weekly_eff_rate_a = total_earnings_a / total_hours_a
weekly_cost_per_unit_a = total_earnings_a / total_units_a
print(f"Total: Units={total_units_a}, Hours={total_hours_a}, Earnings=${total_earnings_a:.2f}, Eff Hourly Rate=${weekly_eff_rate_a:.3f}, Cost/Unit=${weekly_cost_per_unit_a:.3f}")
